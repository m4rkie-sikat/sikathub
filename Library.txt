local function Create(className, props, themeKey, themeProp)
	local inst = Instance.new(className)
	for k, v in pairs(props or {}) do
		inst[k] = v
	end
	if themeKey and themeProp then
		inst[themeProp] = UILibrary.Themes[UILibrary.CurrentTheme][themeKey]
		table.insert(UILibrary.ThemeObjects, { Instance = inst, Key = themeKey, Property = themeProp })
	end
	if inst:IsA("TextLabel") or inst:IsA("TextButton") or inst:IsA("TextBox") then
		if not (props and props.Font) then
			inst.Font = UILibrary.CurrentFont
		end
		table.insert(UILibrary.FontObjects, inst)
	end
	return inst
end

local function MakeDraggable(handle, target)
	local dragging, dragInput, dragStart, startPos
	handle.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = target.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then
					dragging = false
				end
			end)
		end
	end)
	handle.InputChanged:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
			dragInput = input
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if input == dragInput and dragging then
			local delta = input.Position - dragStart
			target.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
		end
	end)
end

function UILibrary:SetTheme(themeName)
	if not self.Themes[themeName] then return end
	self.CurrentTheme = themeName
	local palette = self.Themes[themeName]
	for _, item in ipairs(self.ThemeObjects) do
		if item.Instance and item.Instance.Parent then
			TweenService:Create(item.Instance, TweenInfo.new(0.2), { [item.Property] = palette[item.Key] }):Play()
		end
	end
end

function UILibrary:SetFont(fontEnum)
	self.CurrentFont = fontEnum
	for _, inst in ipairs(self.FontObjects) do
		if inst and inst.Parent then
			inst.Font = fontEnum
		end
	end
end

function UILibrary:SaveConfig(folderName, configName)
	if not (writefile and makefolder and isfolder) then return false end
	if not isfolder(folderName) then makefolder(folderName) end

	local data = {}
	for flag, val in pairs(self.Flags) do
		if typeof(val) == "Color3" then
			data[flag] = { Type = "Color3", R = val.R, G = val.G, B = val.B }
		elseif typeof(val) == "EnumItem" then
			data[flag] = { Type = "EnumItem", Name = val.Name }
		else
			data[flag] = val
		end
	end
	writefile(folderName .. "/" .. configName .. ".json", HttpService:JSONEncode(data))
	return true
end

function UILibrary:LoadConfig(folderName, configName)
	if not (readfile and isfile) then return false end
	local path = folderName .. "/" .. configName .. ".json"
	if not isfile(path) then return false end

	local success, decoded = pcall(function() return HttpService:JSONDecode(readfile(path)) end)
	if not success or type(decoded) ~= "table" then return false end

	for flag, val in pairs(decoded) do
		if self.Elements[flag] and self.Elements[flag].Set then
			if type(val) == "table" and val.Type == "Color3" then
				self.Elements[flag]:Set(Color3.new(val.R, val.G, val.B))
			elseif type(val) == "table" and val.Type == "EnumItem" then
				self.Elements[flag]:Set(Enum.KeyCode[val.Name] or Enum.KeyCode.RightShift)
			else
				self.Elements[flag]:Set(val)
			end
		end
	end
	return true
end

					UpdateFromInput(input)

			UILibrary.Flags[flag] = currentKey

			local Frame = Create("Frame", {
				Parent = Page,
				Size = UDim2.new(1, 0, 0, 34)
			}, "Element", "BackgroundColor3")
			Create("UICorner", { CornerRadius = UDim.new(0, 6), Parent = Frame })

			Create("TextLabel", {
				Parent = Frame,
				Size = UDim2.new(0.6, 0, 1, 0),
				Position = UDim2.new(0, 10, 0, 0),
				BackgroundTransparency = 1,
				Text = cfg.Name or "Keybind",
				TextSize = 13,
				TextXAlignment = Enum.TextXAlignment.Left
			}, "Text", "TextColor3")

			local KeyBtn = Create("TextButton", {
				Parent = Frame,
				Size = UDim2.new(0, 90, 0, 24),
				Position = UDim2.new(1, -98, 0.5, -12),
				Text = currentKey.Name,
				TextSize = 12
			}, "TopBar", "BackgroundColor3")
			Create("UICorner", { CornerRadius = UDim.new(0, 4), Parent = KeyBtn })
			KeyBtn.TextColor3 = UILibrary.Themes[UILibrary.CurrentTheme].Accent

			local ElementObj = {}
			function ElementObj:Set(keyEnum)
				currentKey = keyEnum
				UILibrary.Flags[flag] = currentKey
				KeyBtn.Text = currentKey.Name
				if cfg.OnChange then task.spawn(cfg.OnChange, currentKey) end
			end

			KeyBtn.MouseButton1Click:Connect(function()
				listening = true
				KeyBtn.Text = "..."
			end)

			UserInputService.InputBegan:Connect(function(input, gpe)
				if listening and input.UserInputType == Enum.UserInputType.Keyboard then
					listening = false
					ElementObj:Set(input.KeyCode)
				elseif not gpe and input.KeyCode == currentKey then
					if cfg.Callback then task.spawn(cfg.Callback) end
				end
			end)

			UILibrary.Elements[flag] = ElementObj
			return ElementObj
		end

		function TabObj:AddColorPicker(cfg)
			local flag = cfg.Flag or cfg.Name
			local color = cfg.Default or Color3.fromRGB(255, 0, 0)
			local h, s, v = color:ToHSV()
			local opened = false
			UILibrary.Flags[flag] = color

			local Container = Create("Frame", {
				Parent = Page,
				Size = UDim2.new(1, 0, 0, 34),
				ClipsDescendants = true
			}, "Element", "BackgroundColor3")
			Create("UICorner", { CornerRadius = UDim.new(0, 6), Parent = Container })

			local Header = Create("TextButton", {
				Parent = Container,
				Size = UDim2.new(1, 0, 0, 34),
				BackgroundTransparency = 1,
				Text = "  " .. (cfg.Name or "Color"),
				TextSize = 13,
				TextXAlignment = Enum.TextXAlignment.Left
			}, "Text", "TextColor3")

			local Preview = Create("Frame", {
				Parent = Header,
				Size = UDim2.new(0, 36, 0, 20),
				Position = UDim2.new(1, -44, 0.5, -10),
				BackgroundColor3 = color
			})
			Create("UICorner", { CornerRadius = UDim.new(0, 4), Parent = Preview })

			local SVPad = Create("TextButton", {
				Parent = Container,
				Size = UDim2.new(1, -45, 0, 90),
				Position = UDim2.new(0, 10, 0, 40),
				BackgroundColor3 = Color3.fromHSV(h, 1, 1),
				Text = "",
				AutoButtonColor = false
			})
			Create("UIGradient", {
				Parent = SVPad,
				Color = ColorSequence.new(Color3.new(1, 1, 1), Color3.new(1, 1, 1)),
				Transparency = NumberSequence.new(0, 1)
			})
			local DarkOverlay = Create("Frame", {
				Parent = SVPad,
				Size = UDim2.new(1, 0, 1, 0),
				BackgroundColor3 = Color3.new(0, 0, 0),
				BorderSizePixel = 0
			})
			Create("UIGradient", {
				Parent = DarkOverlay,
				Rotation = 90,
				Transparency = NumberSequence.new(1, 0)
			})

			local HueBar = Create("TextButton", {
				Parent = Container,
				Size = UDim2.new(0, 20, 0, 90),
				Position = UDim2.new(1, -28, 0, 40),
				BackgroundColor3 = Color3.new(1, 1, 1),
				Text = "",
				AutoButtonColor = false
			})
			Create("UIGradient", {
				Parent = HueBar,
				Rotation = 90,
				Color = ColorSequence.new({
					ColorSequenceKeypoint.new(0, Color3.fromRGB(255, 0, 0)),
					ColorSequenceKeypoint.new(0.17, Color3.fromRGB(255, 255, 0)),
					ColorSequenceKeypoint.new(0.33, Color3.fromRGB(0, 255, 0)),
					ColorSequenceKeypoint.new(0.5, Color3.fromRGB(0, 255, 255)),
					ColorSequenceKeypoint.new(0.67, Color3.fromRGB(0, 0, 255)),
					ColorSequenceKeypoint.new(0.83, Color3.fromRGB(255, 0, 255)),
					ColorSequenceKeypoint.new(1, Color3.fromRGB(255, 0, 0))
				})
			})

			local ElementObj = {}
			function ElementObj:Set(newColor)
				color = newColor
				h, s, v = color:ToHSV()
				UILibrary.Flags[flag] = color
				Preview.BackgroundColor3 = color
				SVPad.BackgroundColor3 = Color3.fromHSV(h, 1, 1)
				if cfg.Callback then task.spawn(cfg.Callback, color) end
			end

			local draggingSV, draggingHue = false, false
			SVPad.InputBegan:Connect(function(i)
				if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
					draggingSV = true
				end
			end)
			HueBar.InputBegan:Connect(function(i)
				if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
					draggingHue = true
				end
			end)
			UserInputService.InputEnded:Connect(function(i)
				if i.UserInputType == Enum.UserInputType.MouseButton1 or i.UserInputType == Enum.UserInputType.Touch then
					draggingSV, draggingHue = false, false
				end
			end)
			UserInputService.InputChanged:Connect(function(i)
				if i.UserInputType == Enum.UserInputType.MouseMovement or i.UserInputType == Enum.UserInputType.Touch then
					if draggingSV then
						s = math.clamp((i.Position.X - SVPad.AbsolutePosition.X) / SVPad.AbsoluteSize.X, 0, 1)
						v = 1 - math.clamp((i.Position.Y - SVPad.AbsolutePosition.Y) / SVPad.AbsoluteSize.Y, 0, 1)
						ElementObj:Set(Color3.fromHSV(h, s, v))
					elseif draggingHue then
						h = math.clamp((i.Position.Y - HueBar.AbsolutePosition.Y) / HueBar.AbsoluteSize.Y, 0, 1)
						ElementObj:Set(Color3.fromHSV(h, s, v))
					end
				end
			end)

			Header.MouseButton1Click:Connect(function()
				opened = not opened
				TweenService:Create(Container, TweenInfo.new(0.15), { Size = UDim2.new(1, 0, 0, opened and 140 or 34) }):Play()
			end)

			UILibrary.Elements[flag] = ElementObj
			return ElementObj
		end

		return TabObj
	end

	return WindowObj
end

return UILibrary
